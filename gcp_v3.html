Configurazione Firebase richiesta

Per le lobby e il roster condiviso, sostituisci le regole del Realtime Database con queste e pubblicale. Il codice stanza autorizza chiunque lo conosca, quindi condividilo solo con i partecipanti; il numero di posti viene controllato dall’app:

{
  "rules": {
    "rooms": {
      "$room": {
        "hostUid": {
          ".write": "auth != null && ((!data.exists() && newData.val() === auth.uid) || (data.val() === auth.uid && root.child('rooms').child($room).child('members').child(newData.val()).val() === true))"
        },
        "members": {
          "$uid": {
            ".write": "auth != null && $uid === auth.uid && root.child('rooms').child($room).child('hostUid').exists() && (newData.val() === true || !newData.exists())"
          }
        },
        "lobby": {
          ".read": "auth != null && root.child('rooms').child($room).child('members').child(auth.uid).val() === true",
          ".write": "auth != null && root.child('rooms').child($room).child('members').child(auth.uid).val() === true"
        },
        "state": {
          ".read": "auth != null && root.child('rooms').child($room).child('members').child(auth.uid).val() === true",
          ".write": "auth != null && root.child('rooms').child($room).child('members').child(auth.uid).val() === true"
        }
      }
    }
  }
}
